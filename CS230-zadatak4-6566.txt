
import socket
import os
import time
import datetime

SRV_B_HOST   = "127.0.0.1"
SRV_B_PORT   = 9001
LOCAL_FILE   = "data.txt"
CHECK_INTERVAL = 10


def ts():
    return datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def get_local_size() -> int:
    
    try:
        return os.path.getsize(LOCAL_FILE)
    except FileNotFoundError:
        print(f"[A] [{ts()}] Lokalni fajl '{LOCAL_FILE}' ne postoji!")
        return -1


def ask_remote_size() -> int:
  
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(5)
            s.connect((SRV_B_HOST, SRV_B_PORT))
            s.sendall(b"GET_SIZE\n")
            s.shutdown(socket.SHUT_WR)
            data = s.recv(64).decode("utf-8").strip()
            return int(data)
    except (ConnectionRefusedError, TimeoutError):
        print(f"[A] [{ts()}] Server B nije dostupan.")
        return -1
    except ValueError:
        print(f"[A] [{ts()}] Neispravan odgovor od Servera B.")
        return -1


def notify_mismatch(local_size: int, remote_size: int):
    """Šalje poruku Serveru B o razlici u veličini fajla."""
    msg = (
        f"MISMATCH local={local_size}B remote={remote_size}B "
        f"diff={abs(local_size - remote_size)}B\n"
    )
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(5)
            s.connect((SRV_B_HOST, SRV_B_PORT))
            s.sendall(msg.encode("utf-8"))
        print(f"[A] [{ts()}] Razlika detektovana i prijavljena Serveru B.")
    except OSError as e:
        print(f"[A] [{ts()}] Greška pri slanju MISMATCH poruke: {e}")


def check_once():
 
    local_size  = get_local_size()
    remote_size = ask_remote_size()

    if local_size == -1 or remote_size == -1:
        return

    print(
        f"[A] [{ts()}] Provjera → lokalno={local_size}B  "
        f"remote={remote_size}B"
    )

    if local_size != remote_size:
        notify_mismatch(local_size, remote_size)
    else:
        print(f"[A] [{ts()}] Fajlovi su iste veličine — OK.")


def run():
    print(f"[A] Server A pokrenut. Provjera svakih {CHECK_INTERVAL}s.")
    while True:
        check_once()
        time.sleep(CHECK_INTERVAL)


if __name__ == "__main__":
    run()


server
import socket
import threading
import os
import datetime

HOST       = "127.0.0.1"
PORT       = 9001
LOCAL_FILE = "data.txt"
LOG_FILE   = "replication.log"

log_lock = threading.Lock()


def ts():
    return datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")


def write_log(entry: str):
    
    line = f"[{ts()}] {entry}\n"
    with log_lock:
        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(line)
    print(f"[B] {line.strip()}")


def get_local_size() -> int:
    try:
        return os.path.getsize(LOCAL_FILE)
    except FileNotFoundError:
        return -1


def handle(conn: socket.socket, addr: tuple):
    with conn:
        try:
            data = conn.recv(256).decode("utf-8").strip()

            if data == "GET_SIZE":
                size = get_local_size()
                conn.sendall(f"{size}\n".encode("utf-8"))
                print(f"[B] GET_SIZE od {addr} → {size}B")

            elif data.startswith("MISMATCH"):
                write_log(f"UPOZORENJE od Server A: {data}")

            else:
                print(f"[B] Nepoznata poruka od {addr}: '{data}'")

        except OSError as e:
            print(f"[B] Greška: {e}")


def run():
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as srv:
        srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        srv.bind((HOST, PORT))
        srv.listen()
        print(f"[B] Server B aktivan na {HOST}:{PORT}  |  log → {LOG_FILE}")
        while True:
            conn, addr = srv.accept()
            threading.Thread(
                target=handle,
                args=(conn, addr),
                daemon=True
            ).start()


if __name__ == "__main__":
    run()
