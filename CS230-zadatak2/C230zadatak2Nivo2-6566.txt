import socket
import threading
import itertools
from config import LB_HOST, LB_PORT, SERVERS

pool = itertools.cycle(SERVERS)
pool_lock = threading.Lock()


def next_server():
    with pool_lock:
        return next(pool)


def pipe(src: socket.socket, dst: socket.socket):
    
    try:
        while chunk := src.recv(4096):
            dst.sendall(chunk)
    except OSError:
        pass
    finally:
        for s in (src, dst):
            try: s.shutdown(socket.SHUT_WR)
            except OSError: pass


def handle(client_sock: socket.socket, client_addr: tuple):
    target_host, target_port = next_server()
    print(f"[LB] {client_addr} → {target_host}:{target_port}")
    try:
        srv_sock = socket.create_connection((target_host, target_port), timeout=5)
    except ConnectionRefusedError:
        print(f"[LB] Server {target_port} nedostupan!")
        client_sock.close()
        return

    t1 = threading.Thread(target=pipe, args=(client_sock, srv_sock), daemon=True)
    t2 = threading.Thread(target=pipe, args=(srv_sock, client_sock), daemon=True)
    t1.start(); t2.start()
    t1.join(); t2.join()
    for s in (client_sock, srv_sock):
        try: s.close()
        except OSError: pass


def run():
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as srv:
        srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        srv.bind((LB_HOST, LB_PORT))
        srv.listen()
        print(f"[LB] Sluša na {LB_HOST}:{LB_PORT}  |  pool: {SERVERS}")
        while True:
            conn, addr = srv.accept()
            threading.Thread(target=handle, args=(conn, addr), daemon=True).start()


if __name__ == "__main__":
    run()
